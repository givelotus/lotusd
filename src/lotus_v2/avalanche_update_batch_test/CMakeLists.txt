cmake_minimum_required(VERSION 3.18)
project(lotus_v2_avalanche_update_batch LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_subdirectory(../avalanche_runtime_test runtime)
get_target_property(runtime_sources avalanche_runtime_test SOURCES)
list(REMOVE_ITEM runtime_sources ../test_avalanche_runtime.cpp)
add_executable(avalanche_update_batch_test ${runtime_sources} test_update_batch.cpp)
get_target_property(runtime_includes avalanche_runtime_test INCLUDE_DIRECTORIES)
target_include_directories(avalanche_update_batch_test PRIVATE ${runtime_includes} ../../leveldb)
target_link_libraries(avalanche_update_batch_test PRIVATE lotus_v2_disk_state blake3 OpenSSL::Crypto "${LOTUS_SECP_LIBRARY}")
target_compile_options(avalanche_update_batch_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_test(NAME avalanche_update_batch_test COMMAND avalanche_update_batch_test)
