cmake_minimum_required(VERSION 3.18)
project(lotus_v2_block_preview LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_path(LOTUS_BOOST_INCLUDE boost/multiprecision/cpp_int.hpp PATHS /opt/homebrew/opt/boost/include REQUIRED)
if(NOT DEFINED LOTUS_SECP_LIBRARY OR NOT EXISTS "${LOTUS_SECP_LIBRARY}")
    message(FATAL_ERROR "Set LOTUS_SECP_LIBRARY to locally built ABC libsecp256k1.a")
endif()
get_filename_component(LOTUS_SRC "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
# Borrow the independently owned evaluator without copying/editing dependencies.
set(LOTUS_PREVIEW_CORE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "Directory containing preview block_transition.cpp/h")
get_filename_component(LOTUS_PREVIEW_CORE_SOURCE "${LOTUS_PREVIEW_CORE_SOURCE}" ABSOLUTE)
add_library(preview_blake3 STATIC ../vendor/blake3/blake3.c ../vendor/blake3/blake3_dispatch.c ../vendor/blake3/blake3_portable.c)
target_compile_definitions(preview_blake3 PRIVATE BLAKE3_NO_SSE2 BLAKE3_NO_SSE41 BLAKE3_NO_AVX2 BLAKE3_NO_AVX512 BLAKE3_USE_NEON=0)
add_library(preview_core STATIC "${LOTUS_PREVIEW_CORE_SOURCE}/block_transition.cpp"
    ../output.cpp ../transaction.cpp ../commitment.cpp ../tagged_hash.cpp ../signing.cpp ../../support/cleanse.cpp
    ../vm.cpp ../field_size.cpp ../sparse_state.cpp ../incremental_utxo.cpp ../ordinary_validation.cpp ../monetary.cpp
    ../block.cpp ../block_wire.cpp ../block_cost.cpp ../header_context.cpp ../asert_candidate.cpp ../header.cpp
    ../transaction_pool.cpp
   )
target_include_directories(preview_core PUBLIC "${LOTUS_SRC}" "${LOTUS_SRC}/secp256k1/include" "${LOTUS_BOOST_INCLUDE}" "${LOTUS_PREVIEW_CORE_SOURCE}")
target_compile_options(preview_core PUBLIC -include "${LOTUS_PREVIEW_CORE_SOURCE}/block_transition.h")
target_compile_options(preview_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(preview_core PUBLIC preview_blake3 OpenSSL::Crypto "${LOTUS_SECP_LIBRARY}")
if(LOTUS_SANITIZE)
    target_compile_options(preview_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(preview_core PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
foreach(PREVIEW_CASE IN ITEMS preview complete checksig)
    if(PREVIEW_CASE STREQUAL "preview")
        set(PREVIEW_TEST_SOURCE ../test_block_preview.cpp)
    else()
        set(PREVIEW_TEST_SOURCE ../test_block_${PREVIEW_CASE}.cpp)
    endif()
    add_executable(lotus_v2_block_${PREVIEW_CASE}_test "${PREVIEW_TEST_SOURCE}")
    target_link_libraries(lotus_v2_block_${PREVIEW_CASE}_test PRIVATE preview_core)
    target_compile_options(lotus_v2_block_${PREVIEW_CASE}_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME lotus_v2_block_${PREVIEW_CASE}_test COMMAND lotus_v2_block_${PREVIEW_CASE}_test)
endforeach()
