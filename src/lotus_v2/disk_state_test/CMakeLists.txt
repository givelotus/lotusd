# Isolated native review target using the exact bundled ABC LevelDB/crc32c.
cmake_minimum_required(VERSION 3.18)
project(lotus_v2_disk_state LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake/modules")
include(AddCompilerFlags)
set(LEVELDB_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LEVELDB_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(CRC32C_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CRC32C_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../crc32c crc32c)
add_subdirectory(../../leveldb leveldb)
find_path(LOTUS_BOOST_INCLUDE boost/multiprecision/cpp_int.hpp PATHS /opt/homebrew/opt/boost/include REQUIRED)
add_library(blake3 STATIC ../vendor/blake3/blake3.c ../vendor/blake3/blake3_dispatch.c ../vendor/blake3/blake3_portable.c)
target_compile_definitions(blake3 PRIVATE BLAKE3_NO_SSE2 BLAKE3_NO_SSE41 BLAKE3_NO_AVX2 BLAKE3_NO_AVX512 BLAKE3_USE_NEON=0)
add_library(lotus_v2_mdbx STATIC ../vendor/libmdbx/mdbx.c)
set_target_properties(lotus_v2_mdbx PROPERTIES C_STANDARD 11 POSITION_INDEPENDENT_CODE ON)
target_include_directories(lotus_v2_mdbx PUBLIC ../vendor/libmdbx)
target_compile_definitions(lotus_v2_mdbx PUBLIC MDBX_BUILD_SHARED_LIBRARY=0 PRIVATE NDEBUG MDBX_BUILD_FLAGS="-O3 -DNDEBUG" MDBX_WITHOUT_MSVC_CRT=0)
target_compile_options(lotus_v2_mdbx PRIVATE -O3 -Wno-unused-parameter -Wno-sign-compare -Wno-cast-align -Wno-gnu -Wno-shadow -Wno-documentation -Wno-implicit-fallthrough)
add_library(lotus_v2_disk_state STATIC ../disk_state.cpp ../candidate_store.cpp)
target_include_directories(lotus_v2_disk_state SYSTEM PRIVATE "${LOTUS_BOOST_INCLUDE}")
target_compile_features(lotus_v2_disk_state PUBLIC cxx_std_20)
target_link_libraries(lotus_v2_disk_state PUBLIC lotus_v2_mdbx leveldb)
target_compile_options(lotus_v2_disk_state PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(disk_state_test ../test_disk_state.cpp ../sparse_state.cpp ../incremental_utxo.cpp ../output.cpp ../header.cpp ../tagged_hash.cpp)
target_compile_features(disk_state_test PRIVATE cxx_std_20)
target_include_directories(disk_state_test SYSTEM PRIVATE "${LOTUS_BOOST_INCLUDE}")
target_link_libraries(disk_state_test PRIVATE lotus_v2_disk_state blake3)
target_compile_options(disk_state_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_test(NAME disk_state_test COMMAND disk_state_test)
