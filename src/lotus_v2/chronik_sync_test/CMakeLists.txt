cmake_minimum_required(VERSION 3.18)
project(lotus_v2_chronik_sync LANGUAGES C CXX)
add_subdirectory(../development_runtime_test runtime)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)

add_executable(chronik_sync_test ../chronik_sync.cpp ../chronik_explorer.cpp ../test_chronik_sync.cpp)
target_compile_features(chronik_sync_test PRIVATE cxx_std_20)
target_include_directories(chronik_sync_test PRIVATE ../.. ../../secp256k1/include "${LOTUS_BOOST_INCLUDE}" ../vendor/blake3)
target_link_libraries(chronik_sync_test PRIVATE development_runtime_review_objects)
target_compile_options(chronik_sync_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_test(NAME chronik_sync_test COMMAND chronik_sync_test)
