cmake_minimum_required(VERSION 3.18)
project(lotus_v2_header_context LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_path(LOTUS_BOOST_INCLUDE boost/multiprecision/cpp_int.hpp PATHS /opt/homebrew/opt/boost/include REQUIRED)
add_library(blake3 STATIC ../vendor/blake3/blake3.c ../vendor/blake3/blake3_dispatch.c ../vendor/blake3/blake3_portable.c)
target_compile_definitions(blake3 PRIVATE BLAKE3_NO_SSE2 BLAKE3_NO_SSE41 BLAKE3_NO_AVX2 BLAKE3_NO_AVX512 BLAKE3_USE_NEON=0)
add_executable(header_context_test ../header.cpp ../tagged_hash.cpp ../asert_candidate.cpp ../header_context.cpp ../test_header_context.cpp)
target_include_directories(header_context_test SYSTEM PRIVATE "${LOTUS_BOOST_INCLUDE}")
target_link_libraries(header_context_test PRIVATE blake3)
target_compile_options(header_context_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_test(NAME header_context_test COMMAND header_context_test)
